**Goal:** 在独立测量工程中对比 Signalsmith Stretch 与纯 Rust `pitch_shift` 的延迟、CPU、实时安全、音质与双端构建可行性，产出可复核的选型结论。

**Tech Stack:** Rust 1.88、`signalsmith-stretch` 0.1.3、`pitch_shift` 2.1.0、`hound` 3、`clap` 4、`serde`/`serde_json` 1。

```rust
use anyhow::{Context, Result};
use hound::{WavSpec, WavWriter};
use std::f32::consts::TAU;
use std::path::Path;

pub fn sine_sweep(sample_rate: u32, seconds: f32, from_hz: f32, to_hz: f32) -> Vec<f32> {
    let frames = (sample_rate as f32 * seconds).round() as usize;
    let mut out = Vec::with_capacity(frames);
    let mut phase = 0.0f32;
    for i in 0..frames {
        let progress = if frames > 1 {
            i as f32 / (frames - 1) as f32
        } else {
            0.0
        };
        let hz = from_hz + (to_hz - from_hz) * progress;
        phase += TAU * hz / sample_rate.max(1) as f32;
        out.push((phase.sin() * 0.5).clamp(-1.0, 1.0));
    }
    out
}

pub fn harmonic_voice(sample_rate: u32, seconds: f32, f0: f32) -> Vec<f32> {
    let frames = (sample_rate as f32 * seconds).round() as usize;
    let formants = [(700.0f32, 1.0f32), (1_200.0, 0.6), (2_600.0, 0.25)];
    let mut out = Vec::with_capacity(frames);
    for i in 0..frames {
        let t = i as f32 / sample_rate.max(1) as f32;
        let mut sample = 0.0f32;
        for harmonic in 1..=40 {
            let hz = f0 * harmonic as f32;
            if hz >= sample_rate as f32 * 0.45 {
                break;
            }
            let mut amplitude = 0.0f32;
            for (center, gain) in formants {
                let width = (center * 0.18).max(60.0);
                let d = (hz - center) / width;
                amplitude += gain * (-0.5 * d * d).exp();
            }
            sample += amplitude * (TAU * hz * t).sin() / harmonic as f32;
        }
        out.push((sample * 0.18).clamp(-1.0, 1.0));
    }
    out
}

pub fn impulse_train(sample_rate: u32, seconds: f32, period_frames: usize) -> Vec<f32> {
    let frames = (sample_rate as f32 * seconds).round() as usize;
    let mut out = vec![0.0f32; frames];
    let period = period_frames.max(1);
    let mut i = 0;
    while i < frames {
        out[i] = 1.0;
        i += period;
    }
    out
}

pub fn write_wav(path: &Path, sample_rate: u32, samples: &[f32]) -> Result<()> {
    if let Some(parent) = path.parent().filter(|p| !p.as_os_str().is_empty()) {
        std::fs::create_dir_all(parent)
            .with_context(|| format!("创建目录 {}", parent.display()))?;
    }
    let spec = WavSpec {
        channels: 1,
        sample_rate,
        bits_per_sample: 16,
        sample_format: hound::SampleFormat::Int,
    };
    let mut writer =
        WavWriter::create(path, spec).with_context(|| format!("创建 WAV {}", path.display()))?;
    for &sample in samples {
        let value = (sample.clamp(-1.0, 1.0) * i16::MAX as f32).round() as i16;
        writer.write_sample(value)?;
    }
    writer.finalize()?;
    Ok(())
}
```

```rust
use std::alloc::{GlobalAlloc, Layout, System};
use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering};
use std::time::Duration;

pub struct CountingAllocator;

static ALLOC_COUNT: AtomicUsize = AtomicUsize::new(0);
static MEASURING: AtomicBool = AtomicBool::new(false);

unsafe impl GlobalAlloc for CountingAllocator {
    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
        if MEASURING.load(Ordering::Relaxed) {
            ALLOC_COUNT.fetch_add(1, Ordering::Relaxed);
        }
        unsafe { System.alloc(layout) }
    }

    unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
        unsafe { System.dealloc(ptr, layout) }
    }

    unsafe fn realloc(&self, ptr: *mut u8, layout: Layout, new_size: usize) -> *mut u8 {
        if MEASURING.load(Ordering::Relaxed) {
            ALLOC_COUNT.fetch_add(1, Ordering::Relaxed);
        }
        unsafe { System.realloc(ptr, layout, new_size) }
    }
}

pub fn measure_allocations<F: FnOnce()>(f: F) -> usize {
    ALLOC_COUNT.store(0, Ordering::SeqCst);
    MEASURING.store(true, Ordering::SeqCst);
    f();
    MEASURING.store(false, Ordering::SeqCst);
    ALLOC_COUNT.load(Ordering::SeqCst)
}

#[derive(Debug, Clone, Copy, Default)]
pub struct TimingStats {
    pub p50_ms: f64,
    pub p99_ms: f64,
    pub max_ms: f64,
}

pub fn timing_stats(samples: &[Duration]) -> TimingStats {
    if samples.is_empty() {
        return TimingStats::default();
    }
    let mut values: Vec<f64> = samples.iter().map(|d| d.as_secs_f64() * 1000.0).collect();
    values.sort_by(|a, b| a.total_cmp(b));
    let p50 = values[(values.len() - 1) * 50 / 100];
    let p99 = values[(values.len() - 1) * 99 / 100];
    TimingStats {
        p50_ms: p50,
        p99_ms: p99,
        max_ms: *values.last().unwrap(),
    }
}
```

```rust
pub trait Backend {
    fn name(&self) -> &'static str;
    fn process(&mut self, input: &[f32], output: &mut [f32]) -> usize;
    fn latency_frames(&self) -> usize;
    fn reset(&mut self);
}

pub struct PassthroughBackend;

impl Backend for PassthroughBackend {
    fn name(&self) -> &'static str {
        "passthrough"
    }

    fn process(&mut self, input: &[f32], output: &mut [f32]) -> usize {
        let n = input.len().min(output.len());
        output[..n].copy_from_slice(&input[..n]);
        n
    }

    fn latency_frames(&self) -> usize {
        0
    }

    fn reset(&mut self) {}
}

pub struct DelayBackend {
    delay: usize,
    history: Vec<f32>,
    pos: usize,
}

impl DelayBackend {
    pub fn new(delay: usize) -> Self {
        Self {
            delay,
            history: vec![0.0; delay.max(1)],
            pos: 0,
        }
    }
}

impl Backend for DelayBackend {
    fn name(&self) -> &'static str {
        "delay"
    }

    fn process(&mut self, input: &[f32], output: &mut [f32]) -> usize {
        let n = input.len().min(output.len());
        for i in 0..n {
            let delayed = self.history[self.pos];
            self.history[self.pos] = input[i];
            self.pos = (self.pos + 1) % self.history.len();
            output[i] = delayed;
        }
        n
    }
```

```rust
use crate::backend::Backend;
use std::collections::VecDeque;

pub struct BlockAdapter<B: Backend> {
    backend: B,
    block_frames: usize,
    input: Vec<f32>,
    input_len: usize,
    output: VecDeque<f32>,
    wet: Vec<f32>,
    overflow: u64,
}

impl<B: Backend> BlockAdapter<B> {
    pub fn new(backend: B, block_frames: usize) -> Self {
        let block_frames = block_frames.max(1);
        Self {
            backend,
            block_frames,
            input: vec![0.0; block_frames],
            input_len: 0,
            output: VecDeque::with_capacity(block_frames * 8),
            wet: vec![0.0; block_frames],
            overflow: 0,
        }
    }

    pub fn process(&mut self, input: &[f32], output: &mut [f32]) -> usize {
        let mut produced = 0;
        for &sample in input {
            self.input[self.input_len] = sample;
            self.input_len += 1;
            if self.input_len == self.block_frames {
                self.backend.process(&self.input, &mut self.wet);
                for &sample in &self.wet {
                    if self.output.len() == self.output.capacity() {
                        self.output.pop_front();
                        self.overflow += 1;
                    }
                    self.output.push_back(sample);
                }
                self.input_len = 0;
            }
            while produced < output.len() {
                let Some(sample) = self.output.pop_front() else {
                    break;
                };
                output[produced] = sample;
                produced += 1;
            }
        }
        produced
    }

    pub fn overflow_count(&self) -> u64 {
        self.overflow
    }

    pub fn latency_frames(&self) -> usize {
        self.backend.latency_frames() + self.block_frames
    }
}
```

```rust
    pub fn new(

        block_ms: f32,
        semitones: f32,
        formant_semitones: f32,
    ) -> Self {
        let block_frames = ((sample_rate as f32 * block_ms / 1000.0).round() as usize).max(64);

            semitones,
            formant_semitones,
        }
    }

        self.block_frames
    }

        self.formant_semitones
    }

        self.semitones = semitones;
        self.formant_semitones = formant_semitones;
        self.stretch
            .set_transpose_factor_semitones(semitones, None);
        self.stretch
            .set_formant_factor_semitones(formant_semitones, false);
    }
}

    fn name(&self) -> &'static str {
        "signalsmith"
    }

    }

    }
```

```rust
        let state: Box<RawState> = Box::new([0.0; pitch_shift::TOTAL_F32]);

        n
    }

        self.latency_frames
    }

    fn reset(&mut self) {
        let state: Box<RawState> = Box::new([0.0; pitch_shift::TOTAL_F32]);
        self.shifter = Shifter::new(state);
    }
```

```rust
fn backend_block_frames(cfg: &CaseConfig) -> usize {
    match cfg.choice {
        BackendChoice::PitchShift => 128,
        _ => ((cfg.sample_rate as f32 * cfg.block_ms / 1000.0).round() as usize).max(1),

fn make_backend(cfg: &CaseConfig) -> Box<dyn Backend> {
    match cfg.choice {
        BackendChoice::Signalsmith => Box::new(SignalsmithBackend::new(
            cfg.sample_rate,
            cfg.block_ms,
            cfg.semitones,
            cfg.formant_semitones,
        )),
        BackendChoice::PitchShift => {
            Box::new(PitchShiftBackend::new(cfg.sample_rate, cfg.semitones))
        }
        BackendChoice::Passthrough => Box::new(crate::backend::PassthroughBackend),
    }

fn measure_impulse_latency(cfg: &CaseConfig) -> Option<usize> {
    let block = backend_block_frames(cfg);
    let mut adapter = BlockAdapter::new(make_backend(cfg), block);
    let mut input = vec![0.0f32; cfg.chunk_frames.max(1)];
    input[0] = 1.0;
    let mut output = vec![0.0f32; cfg.chunk_frames.max(1)];
    let mut consumed = 0usize;
    let mut output_index = 0usize;
    let max_frames = cfg.sample_rate as usize * 2;
    while consumed < max_frames {
        output.fill(0.0);
        let produced = adapter.process(&input, &mut output);
        if let Some(pos) = output[..produced].iter().position(|s| s.abs() > 1e-6) {
            // 输出流里的位置 + 首次出样本的块缓冲延迟。
            return Some(output_index + pos + block);
        }
        output_index += produced;
        consumed += input.len();
        input.fill(0.0);
    }
    None
}

pub fn run_case(cfg: &CaseConfig) -> Result<CaseResult> {
    let signal = harmonic_voice(cfg.sample_rate, cfg.seconds, 220.0);
    let block = backend_block_frames(cfg);
    let backend = make_backend(cfg);
    let latency_backend = backend.latency_frames();
    let mut adapter = BlockAdapter::new(backend, block);
    let latency_adapter = adapter.latency_frames();
    let chunk = cfg.chunk_frames.max(1);
    let mut out = vec![0.0f32; chunk];
    let mut durations = Vec::with_capacity(signal.len().div_ceil(chunk));
    let mut underruns = 0u64;
    let mut output_frames = 0usize;
    let warmup_chunks = block.div_ceil(chunk) + 2;
    let started = Instant::now();
    let allocations = measure_allocations(|| {
        for (index, input) in signal.chunks(chunk).enumerate() {
            out.fill(0.0);
            let start = Instant::now();
            let produced = adapter.process(input, &mut out);
            durations.push(start.elapsed());
            output_frames += produced;
            if index >= warmup_chunks && produced < input.len() {
                underruns += 1;
            }
        }
    });
    let elapsed = started.elapsed().as_secs_f64();
    let stats = timing_stats(&durations);
    Ok(CaseResult {
        backend: cfg.choice_name().to_string(),
        block_ms: cfg.block_ms,
        chunk_frames: chunk,
        sample_rate: cfg.sample_rate,
        seconds: cfg.seconds,
        input_frames: signal.len(),
        output_frames,
        latency_backend_frames: latency_backend,
        latency_adapter_frames: latency_adapter,
        latency_measured_frames: measure_impulse_latency(cfg),
        p50_ms: stats.p50_ms,
        p99_ms: stats.p99_ms,
        max_ms: stats.max_ms,
        realtime_factor: output_frames as f64 / (elapsed * cfg.sample_rate as f64),
        allocations,
        underruns,
    })
}
```

```rust
pub fn midi_ratio(semitones: f32) -> f32 {
    2.0f32.powf(semitones / 12.0)
}

pub fn estimate_frequency(samples: &[f32], sample_rate: u32) -> f32 {
    if samples.len() < 2 {
        return 0.0;
    }
    let mut crossings = 0usize;
    let mut previous = samples[0];
    for &sample in &samples[1..] {
        if (previous <= 0.0 && sample > 0.0) || (previous >= 0.0 && sample < 0.0) {
            crossings += 1;
        }
        previous = sample;
    }
    let seconds = samples.len() as f32 / sample_rate.max(1) as f32;
    crossings as f32 / (2.0 * seconds)
}
```
